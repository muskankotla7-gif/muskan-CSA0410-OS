#include <stdio.h>

int main() {
    int n, i, time = 0, completed = 0;
    int bt[20], rt[20], pr[20];
    int wt[20] = {0}, tat[20];
    int highest, minPriority;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("Enter Burst Time and Priority of P%d: ", i + 1);
        scanf("%d%d", &bt[i], &pr[i]);
        rt[i] = bt[i];
    }

    while(completed < n) {
        highest = -1;
        minPriority = 9999;

        for(i = 0; i < n; i++) {
            if(rt[i] > 0 && pr[i] < minPriority) {
                minPriority = pr[i];
                highest = i;
            }
        }

        if(highest == -1) {
            time++;
            continue;
        }

        rt[highest]--;
        time++;

        if(rt[highest] == 0) {
            completed++;
            tat[highest] = time;
            wt[highest] = tat[highest] - bt[highest];
        }
    }

    printf("\nProcess\tBT\tPR\tWT\tTAT\n");
    for(i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\t%d\n",
               i + 1, bt[i], pr[i], wt[i], tat[i]);
    }

    return 0;
}
